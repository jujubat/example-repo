  - active (boolean)
  - created_at (timestamp)

  - destinations (list)
  - routes (list with scores)
  - weather (dict)
  - timestamp (timestamp)

alerts
  - type (slack|whatsapp)
  - recipient (string)
  - message (string)
  - status (sent|failed)
  - timestamp (timestamp)

weather
  - location (string)
  - data (weather details)
  - timestamp (timestamp)

logs
  - timestamp (timestamp)
  - message (string)
  - level (INFO|WARNING|ERROR)
